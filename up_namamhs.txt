print("===================================")
print("       SISTEM PENILAIAN AKADEMIK")
print("===================================")

jumlah = int(input("Masukkan jumlah mahasiswa: "))

lulus = 0
tidak_lulus = 0

nama_mahasiswa = []
hasil_penilaian = []

for i in range(jumlah):
    print("\nMahasiswa ke-", i + 1)

    nama = input("Masukkan nama mahasiswa: ")
    nilai = float(input("Masukkan nilai mahasiswa: "))

    if nilai < 0 or nilai > 100:
        print("Nilai tidak valid!")
        continue

    if nilai >= 85:
        kategori = "A (Sangat Baik)"
        lulus += 1
    elif nilai >= 75:
        kategori = "B (Baik)"
        lulus += 1
    elif nilai >= 60:
        kategori = "C (Cukup)"
        lulus += 1
    else:
        kategori = "D (Tidak Lulus)"
        tidak_lulus += 1

    nama_mahasiswa.append(nama)
    hasil_penilaian.append((nilai, kategori))

print("\n===================================")
print("          HASIL PENILAIAN")
print("===================================")

for i in range(len(nama_mahasiswa)):
    print("Nama  :", nama_mahasiswa[i])
    print("Nilai :", hasil_penilaian[i][0])
    print("Hasil :", hasil_penilaian[i][1])
    print("-----------------------------------")

print("Jumlah mahasiswa lulus       :", lulus)
print("Jumlah mahasiswa tidak lulus :", tidak_lulus)
print("===================================")
